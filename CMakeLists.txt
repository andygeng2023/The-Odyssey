cmake_minimum_required(VERSION 3.20)
project(odyssey_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_EXTERNAL_GLFW OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(raylib)
file(GLOB_RECURSE ODYSSEY_SOURCES CONFIGURE_DEPENDS engine/*.cpp engine/*.h)
add_executable(odyssey ${ODYSSEY_SOURCES})
target_link_libraries(odyssey PRIVATE raylib)
target_compile_definitions(odyssey PRIVATE ODYSSEY_ENGINE_VERSION="0.2.0")
if(MSVC)
 target_compile_options(odyssey PRIVATE /W4 /permissive-)
else()
 target_compile_options(odyssey PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
